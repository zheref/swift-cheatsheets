Map for dictionaries, objects for records

A plain object inherits keys like `__proto__`; a Map has none and keeps insertion order.

const counts = new Map<string, number>();
for (const word of "the cat and the hat".split(" ")) {
  counts.set(word, (counts.get(word) ?? 0) + 1);
}
console.log([...counts]);                    // insertion order
console.log(counts.get("the"), counts.has("dog"));

const record: Record<string, number> = {};
record["__proto__"] = 1;                     // hits the inherited __proto__ setter: no key is added

Pitfalls

Where code that reads like Swift behaves differently.

  • queue.shift() in a loop is O(n²) overall — keep a head index or use a ring buffer.
  • delete xs[i] leaves a hole (length unchanged); splice(i, 1) removes.
  • [1, 2] === [1, 2] is false: arrays and objects compare by reference, also as Set/Map keys.
  • JSON.stringify(new Map()) is {} — serialise [...map] or Object.fromEntries(map).
  • Use Map for anything keyed at runtime, Set for membership, and readonly in signatures.
  • Arrays have no value semantics: const ys = xs aliases. Copy with [...xs] or structuredClone(xs).

Chunk, window and dedupe

Not built in — two small generic helpers cover both.

function chunk<T>(xs: readonly T[], size: number): T[][] {
  return Array.from({ length: Math.ceil(xs.length / size) }, (_, i) => xs.slice(i * size, i * size + size));
}
function windows<T>(xs: readonly T[], size: number): T[][] {
  return xs.slice(0, Math.max(0, xs.length - size + 1)).map((_, i) => xs.slice(i, i + size));
}

Sequential vs concurrent

A promise starts when it is created; `await` only waits. Start both, then await both.

const sleep = (ms: number) => new Promise<void>((resolve) => setTimeout(resolve, ms));
async function load(name: string, ms: number): Promise<string> {
  await sleep(ms);
  return name;
}

let start = performance.now();
const a = await load("user", 100);           // total = a + b (about 200 ms)
const b = await load("posts", 100);
const sequential = performance.now() - start;

start = performance.now();
const [c, d] = await Promise.all([load("user", 100), load("posts", 100)]);   // total = max(a, b) (about 100 ms)
const concurrent = performance.now() - start;
console.log(a, b, "|", c, d);
console.log("concurrent took well under the sequential time:", concurrent < sequential * 0.75);

Cancellation with AbortSignal

Promises cannot be cancelled; work is. Pass a signal down, honour it at every await, and detach the listener when the wait ends normally.

function delay(ms: number, signal?: AbortSignal): Promise<void> {
  return new Promise((resolve, reject) => {
    if (signal?.aborted) return reject(signal.reason);
    const onAbort = () => {
      clearTimeout(timer);
      reject(signal!.reason);
    };
    const timer = setTimeout(() => {
      signal?.removeEventListener("abort", onAbort);   // no listener left behind per tick
      resolve();
    }, ms);
    signal?.addEventListener("abort", onAbort, { once: true });
  });
}

async function poll(signal: AbortSignal, onTick: (i: number) => void) {
  for (let i = 1; ; i++) {
    await delay(50, signal);                 // each await is a cancellation point
    console.log("tick", i);
    onTick(i);
  }
}

const controller = new AbortController();
try {
  await poll(controller.signal, (i) => {
    if (i === 2) controller.abort();          // e.g. the user left the screen
  });
} catch (e) {
  console.log((e as DOMException).name);     // AbortSignal.timeout(ms) rejects the same way, as TimeoutError
}

Rules

What reviewers flag first.

  • A floating promise (no await, no return, no .catch) loses its errors — Node then exits on the unhandled rejection.
  • Independent work: Promise.all. Partial failure acceptable: Promise.allSettled.
  • Accept an AbortSignal in every long-running function and pass it to fetch, timers and streams.
  • AbortSignal.timeout(ms) and AbortSignal.any([a, b]) compose deadlines without timers of your own.
  • CPU-bound work blocks everything: move it to a Worker (browser) or worker_threads (Node).

Two pointers

On sorted input, move the pointer that brings the sum closer. O(n), O(1) space.

function pairWithSum(sorted: readonly number[], target: number): [number, number] | null {
  let lo = 0;
  let hi = sorted.length - 1;
  while (lo < hi) {
    const sum = sorted[lo]! + sorted[hi]!;
    if (sum === target) return [lo, hi];
    if (sum < target) lo++;
    else hi--;
  }
  return null;
}

Sliding window

Longest substring without repeats — grow the right edge, jump the left past the last repeat. O(n).

function longestUnique(s: string): number {
  const lastSeen = new Map<string, number>();
  let start = 0;
  let best = 0;
  for (let i = 0; i < s.length; i++) {
    const prev = lastSeen.get(s[i]!);
    if (prev !== undefined && prev >= start) start = prev + 1;
    lastSeen.set(s[i]!, i);
    best = Math.max(best, i - start + 1);
  }
  return best;
}

Binary search — lower bound

The first index whose value is not less than the target. Plain division keeps the midpoint exact — `>>> 1` would wrap past 2^31 elements.

function lowerBound(xs: readonly number[], target: number): number {
  let lo = 0;
  let hi = xs.length;                        // the answer lies in [lo, hi]
  while (lo < hi) {
    const mid = Math.floor((lo + hi) / 2);   // exact up to 2^53
    if (xs[mid]! < target) lo = mid + 1;
    else hi = mid;
  }
  return lo;
}

Heap — top k

Keep a min-heap of size k; its root is the smallest of the k largest. O(n log k).

class MinHeap {
  private items: number[] = [];
  get size() { return this.items.length; }
  push(x: number) {
    const a = this.items;
    a.push(x);
    for (let i = a.length - 1; i > 0; ) {
      const p = (i - 1) >> 1;
      if (a[p]! <= a[i]!) break;
      [a[p], a[i]] = [a[i]!, a[p]!];
      i = p;
    }
  }
  pop(): number | undefined {
    const a = this.items;
    const top = a[0];
    const last = a.pop();
    if (a.length && last !== undefined) {
      a[0] = last;
      for (let i = 0; ; ) {
        const l = 2 * i + 1;
        const r = l + 1;
        let m = i;
        if (l < a.length && a[l]! < a[m]!) m = l;
        if (r < a.length && a[r]! < a[m]!) m = r;
        if (m === i) break;
        [a[m], a[i]] = [a[i]!, a[m]!];
        i = m;
      }
    }
    return top;
  }
}

function topK(xs: readonly number[], k: number): number[] {
  const heap = new MinHeap();
  for (const x of xs) {
    heap.push(x);
    if (heap.size > k) heap.pop();
  }
  const out: number[] = [];
  while (heap.size) out.push(heap.pop()!);
  return out.reverse();
}

Breadth-first search on a grid

BFS finds shortest paths in unweighted graphs. Dequeue by index — `shift()` would make it O(n²).

function shortestPath(grid: string[], from: [number, number], to: [number, number]): number {
  const rows = grid.length;
  const cols = grid[0]!.length;
  const dist = new Map<string, number>([[from.join(","), 0]]);
  const queue: [number, number][] = [from];
  for (let head = 0; head < queue.length; head++) {
    const [r, c] = queue[head]!;
    const here = dist.get(`${r},${c}`)!;
    if (r === to[0] && c === to[1]) return here;
    for (const [dr, dc] of [[1, 0], [-1, 0], [0, 1], [0, -1]] as const) {
      const nr = r + dr;
      const nc = c + dc;
      const key = `${nr},${nc}`;
      if (nr < 0 || nc < 0 || nr >= rows || nc >= cols || grid[nr]![nc] === "#" || dist.has(key)) continue;
      dist.set(key, here + 1);
      queue.push([nr, nc]);
    }
  }
  return -1;
}

Rules

Make illegal states unrepresentable — within what erasure allows.

  • A discriminated union over a bag of optional fields: { status: "error"; error: E } cannot lack its error.
  • String-literal unions over enum: they erase completely (Node's type stripping rejects enum).
  • as is an assertion, not a conversion — x as User on unchecked data is a bug waiting.
  • any switches checking off for everything it touches; unknown forces a check.
  • Types vanish at runtime: anything crossing a boundary (network, storage, user input) needs validation.

Writing atomically

Write a temporary file in the same directory, flush it, then rename over the target: readers see the old file or the new one, never half.

import { randomUUID } from "node:crypto";
import { mkdtemp, open, readFile, rename, rm } from "node:fs/promises";
import { tmpdir } from "node:os";
import { basename, dirname, join } from "node:path";

async function writeFileAtomic(path: string, data: string): Promise<void> {
  const temp = join(dirname(path), `.${basename(path)}.${randomUUID()}.tmp`);   // same file system, unique per call
  const handle = await open(temp, "wx");     // exclusive: never truncates another writer's file
  try {
    await handle.writeFile(data, "utf8");
    await handle.sync();                     // on the device before the rename publishes it
  } finally {
    await handle.close();
  }
  await rename(temp, path);                  // atomic replace on POSIX file systems
}

const dir = await mkdtemp(join(tmpdir(), "atomic-"));
const settings = join(dir, "settings.json");
await writeFileAtomic(settings, JSON.stringify({ theme: "dark" }));
await writeFileAtomic(settings, JSON.stringify({ theme: "light" }));
console.log(JSON.parse(await readFile(settings, "utf8")).theme);
await rm(dir, { recursive: true });

Rules that prevent data loss

Most file bugs are about the moment between two writes.

  • Replace a file by writing a temporary sibling and rename-ing it; never truncate the original first.
  • await every write and close handles in finally; a crash between writes is the case to design for.
  • readFileSync on a request path blocks every other client of the process.
  • On Windows rename over an open file can fail with EPERM/EBUSY — retry, or keep readers short-lived.
  • fs.watch events are hints, not a log: re-read the file instead of trusting the event payload.

A store with EventTarget

One owner of the state, replaced (never mutated) on every update, and a change event to re-render from. `EventTarget` exists in Node too, so this one runs there.

type State = { items: string[]; filter: string };

class Store extends EventTarget {
  #state: State = { items: [], filter: "" };
  get state(): Readonly<State> {
    return this.#state;
  }
  update(change: Partial<State>) {
    this.#state = { ...this.#state, ...change };   // a new object: `old !== new` detects change
    this.dispatchEvent(new Event("change"));
  }
}

Cancel superseded requests

Each keystroke aborts the previous search, so a slow old response can never overwrite a newer one.

let inFlight: AbortController | undefined;

async function search(query: string): Promise<string[]> {
  inFlight?.abort();                         // supersede the previous request
  const controller = new AbortController();
  inFlight = controller;
  const response = await fetch(`/api/search?q=${encodeURIComponent(query)}`, {
    signal: AbortSignal.any([controller.signal, AbortSignal.timeout(5_000)]),
  });
  if (!response.ok) throw new Error(`HTTP ${response.status}`);
  return (await response.json()) as string[];
}

const box = document.querySelector<HTMLInputElement>("#search")!;
box.addEventListener("input", async () => {
  try {
    console.log(await search(box.value));
  } catch (e) {
    if (e instanceof DOMException && e.name === "AbortError") return;   // superseded, not a failure
    throw e;
  }
});

Persist preferences safely

Storage can be disabled, full or hold an old shape — read defensively and version the key.

type Prefs = { theme: "light" | "dark" };
const KEY = "prefs:v1";
const DEFAULTS: Prefs = { theme: "light" };

function loadPrefs(): Prefs {
  try {
    const parsed: unknown = JSON.parse(localStorage.getItem(KEY) ?? "null");
    return typeof parsed === "object" && parsed !== null && "theme" in parsed && (parsed.theme === "light" || parsed.theme === "dark")
      ? { theme: parsed.theme }
      : DEFAULTS;
  } catch {
    return DEFAULTS;                         // private mode, quota, corrupt JSON
  }
}

function savePrefs(prefs: Prefs): void {
  try {
    localStorage.setItem(KEY, JSON.stringify(prefs));
  } catch {
    // keep working without persistence
  }
}

const prefs = loadPrefs();
document.documentElement.dataset.theme = prefs.theme;
savePrefs({ theme: prefs.theme === "light" ? "dark" : "light" });
Result

The page takes the stored theme (light on first load) and the next load starts in the other one.
